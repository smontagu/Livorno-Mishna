\tractatehead{מסכת נגעים}{negaim}

\perek{מַרְאוֹת}{נְגָעִים שְׁנַיִם שֶׁהֵן אַרְבָּעָה\hdot
בַּהֶרֶת עַזָּה כַּשֶּׁלֶג\hdot
שְׁנִיָּה לָהּ כְּסִיד הַהֵיכָל\hdot
וְהַשְּׂאֵת כִּקְרוּם בֵּיצָה\hdot
שְׁנִיָּה לָהּ כְּצֶמֶר לָבָן\hdot
דִּבְרֵי רִבִּי מֵאִיר\hdot
וַחֲכָמִים אוֹמְרִים הַשְּׂאֵת כְּצֶמֶר לָבָן\hdot
שְׁנִיָּה לָהּ כִּקְרוּם בֵּיצָה׃}

\halacha{הַפִּתּוּךְ שֶׁבַּשֶּׁלֶג כְּיַיִן הַמָּזוּג בַּשֶּׁלֶג\hdot
הַפִּתּוּךְ שֶׁבַּסִּיד כְּדָם הַמָּזוּג בֶּחָלָב\hdot
דִּבְרֵי רִבִּי יִשְׁמָעֵאל\hdot
רִבִּי עֲקִיבָא אוֹמֵר אֲדַמְדָּם שֶׁבַּזֶּה וְשֶׁבַּזֶּה\hdot
כְּיַיִן הַמָּזוּג בַּמַּיִם אֶלָּא שֶׁל שֶׁלֶג עַזָּה\hdot
וְשֶׁל סִיד דִּיהָה מִמֶּנָּה׃}

\halacha{אַרְבָּעָה מַרְאוֹת הָאֵלּוּ\hdot
מִצְטָֽרְפִין זֶה עִם זֶה\hdot
לִפְטוֹר וּלְהַחְלִיט וּלְהַסְגִּיר\hdot
לְהַסְגִּיר אֶת הָעוֹמֵד בְּסוֹף שָׁבוּעַ רִאשׁוֹן\hdot
לִפְטוֹר אֶת הָעוֹמֵד בְּסוֹף שָׁבוּעַ שֵׁנִי\hdot
לְהַחֲלִיט אֶת שֶׁנּוֹלַד לוֹ מִחְיָה\hdot
אוֹ שֵׂעָר לָבָן\hdot
בַּתְּחִלָּה בְּסוֹף שָׁבוּעַ רִאשׁוֹן\hdot
בְּסוֹף שָׁבוּעַ שֵׁנִי\hdot
לְאַחַר הַפִּטּוּר\hdot
לְהַחֲלִיט אֶת שֶׁנּוֹלַד לוֹ פִשְׂיוֹן\hdot
בְּסוֹף שָׁבוּעַ רִאשׁוֹן\hdot
בְּסוֹף שָׁבוּעַ שֵׁנִי\hdot
לְאַחַר הַפִּטּוּר\hdot
לְהַחֲלִיט אֶת הַהוֹפֵךְ כֻּלּוֹ לָבָן מִתּוֹךְ הַפִּטּוּר\hdot
לִפְטוֹר אֶת הַהוֹפֵךְ כֻּלּוֹ לָבָן מִתּוֹךְ הֶחְלֵט\hdot
אוֹ מִתּוֹךְ הֶסְגֵּר\hdot
אֵלּוּ מַרְאוֹת נְגָעִים\hdot
שֶׁכָּל נְגָעִים תְּלוּיִם בָּהֶן׃}
