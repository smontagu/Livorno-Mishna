\tractatehead{מסכת נגעים}{negaim}

\perek{מַרְאוֹת}{נְגָעִים שְׁנַיִם שֶׁהֵן אַרְבָּעָה\hdot
בַּהֶרֶת עַזָּה כַּשֶּׁלֶג\hdot
שְׁנִיָּה לָהּ כְּסִיד הַהֵיכָל\hdot
וְהַשְּׂאֵת כִּקְרוּם בֵּיצָה\hdot
שְׁנִיָּה לָהּ כְּצֶמֶר לָבָן\hdot
דִּבְרֵי רִבִּי מֵאִיר\hdot
וַחֲכָמִים אוֹמְרִים הַשְּׂאֵת כְּצֶמֶר לָבָן\hdot
שְׁנִיָּה לָהּ כִּקְרוּם בֵּיצָה׃}

\halacha{הַפִּתּוּךְ שֶׁבַּשֶּׁלֶג כְּיַיִן הַמָּזוּג בַּשֶּׁלֶג\hdot
הַפִּתּוּךְ שֶׁבַּסִּיד כְּדָם הַמָּזוּג בֶּחָלָב\hdot
דִּבְרֵי רִבִּי יִשְׁמָעֵאל\hdot
רִבִּי עֲקִיבָא אוֹמֵר אֲדַמְדָּם שֶׁבַּזֶּה וְשֶׁבַּזֶּה\hdot
כְּיַיִן הַמָּזוּג בַּמַּיִם אֶלָּא שֶׁל שֶׁלֶג עַזָּה\hdot
וְשֶׁל סִיד דִּיהָה מִמֶּנָּה׃}

\halacha{אַרְבָּעָה מַרְאוֹת הָאֵלּוּ\hdot
מִצְטָֽרְפִין זֶה עִם זֶה\hdot
לִפְטוֹר וּלְהַחְלִיט וּלְהַסְגִּיר\hdot
לְהַסְגִּיר אֶת הָעוֹמֵד בְּסוֹף שָׁבוּעַ רִאשׁוֹן\hdot
לִפְטוֹר אֶת הָעוֹמֵד בְּסוֹף שָׁבוּעַ שֵׁנִי\hdot
לְהַחֲלִיט אֶת שֶׁנּוֹלַד לוֹ מִחְיָה\hdot
אוֹ שֵׂעָר לָבָן\hdot
בַּתְּחִלָּה בְּסוֹף שָׁבוּעַ רִאשׁוֹן\hdot
בְּסוֹף שָׁבוּעַ שֵׁנִי\hdot
לְאַחַר הַפִּטּוּר\hdot
לְהַחֲלִיט אֶת שֶׁנּוֹלַד לוֹ פִשְׂיוֹן\hdot
בְּסוֹף שָׁבוּעַ רִאשׁוֹן\hdot
בְּסוֹף שָׁבוּעַ שֵׁנִי\hdot
לְאַחַר הַפִּטּוּר\hdot
לְהַחֲלִיט אֶת הַהוֹפֵךְ כֻּלּוֹ לָבָן מִתּוֹךְ הַפִּטּוּר\hdot
לִפְטוֹר אֶת הַהוֹפֵךְ כֻּלּוֹ לָבָן מִתּוֹךְ הֶחְלֵט\hdot
אוֹ מִתּוֹךְ הֶסְגֵּר\hdot
אֵלּוּ מַרְאוֹת נְגָעִים\hdot
שֶׁכָּל נְגָעִים תְּלוּיִם בָּהֶן׃}

\halacha{רִבִּי חֲנִינָא סְגַן הַכֹּהֲנִים אוֹמֵר מַרְאוֹת נְגָעִים שִׁשָּׁה עָשָׂר\hdot
רִבִּי דוֹסָא בֶן הַרְכִּינָס אוֹמֵר מַרְאוֹת נְגָעִים שְׁלֹשִׁים וְשִׁשָּׁה\hdot
עֲקַבְיָא בֶן מַהֲלַלְאֵל אוֹמֵר שִׁבְעִים וּשְׁתַּיִם\hdot
רִבִּי חֲנִינָא סְגַן הַכֹּהֲנִים אוֹמֵר אֵין רוֹאִין אֶת הַנְּגָעִים בַּתְּחִלָּה\hdot
לְאַחַר הַשַּׁבָּת\hdot
שֶׁשָּׁבוּעַ שֶׁלּוֹ חַל לִהְיוֹת בְּשַׁבָּת\hdot
וְלֹא בַשֵּׁנִי שֶׁשָּׁבוּעַ שֵׁנִי שֶׁלּוֹ חַל לִהְיוֹת בְּשַׁבָּת\hdot
וְלֹא בַשְּׁלִישִׁי לְבָתִּים שֶׁשָּׁבוּעַ שְׁלִישִׁי שֶׁלּוֹ חַל לִהְיוֹת בְּשַׁבָּת\hdot
רִבִּי עֲקִיבָא אוֹמֵר לְעוֹלָם רוֹאִים\hdot
חַל לִהְיוֹת בְּתוֹךְ הַשַּׁבָּת\hdot
מַעֲבִירִין לְאַחַר הַשַּׁבָּת\hdot
וְיֵשׁ בַּדָּבָר לְהָקֵל וּלְהַחֲמִיר׃}

\halacha{כֵּיצַד לְהָקֵל\hdot
הָיָה בוֹ שֵׂעָר לָבָן\hdot
וְהָלַךְ לוֹ שֵׂעָר לָבָן\hdot
הָיוּ לְבָנוֹת וְהִשְׁחִירוּ\hdot
אַחַת לְבָנָה וְאַחַת שְׁחוֹרָה וְהִשְׁחִירוּ שְׁתֵּיהֶן\hdot
אֲרוּכוֹת וְהִקְצִירוּ\hdot
אַחַת אֲרֻכָּה וְאַחַת קְצָרָה וְהִקְצִירוּ שְׁתֵּיהֶן\hdot
נִסְמַךְ הִשְּׁחִין לִשְׁתֵּיהֶן אוֹ לְאַחַת מֵהֶן\hdot
הִקִּיף הַשְּׁחִין לִשְׁתֵּיהֶן\hdot
אוֹ לְאַחַת מֵהֶן\hdot
אוֹ חִלְּקָן הַשְּׁחִין\hdot
וּמִחְיַת הַשְּׁחִין וְהַמִּכְוָה\hdot
וּמִחְיַת הַמִּכְוָה\hdot
וְהַבּוֹהַק\hdot
הָֽיְתָה בוֹ מִחְיָה וְהָֽלְכָה לָהּ הַמִּחְיָה\hdot
הָֽיְתָה מְרֻבַּעַת\hdot
וְנַעֲשֵׂית עֲגוּלָה אוֹ אֲרֻכָּה\hdot
מְבֻצֶרֶת וְנַעֲשֵׂית מִן הַצַּד\hdot
מְכֻנֶּסֶת וְנִתְפַּזְּרָה\hdot
וּבָא הַשְּׁחִין וְנִכְנַס בְּתוֹכָהּ\hdot
הִקִּיפָהּ\hdot
חִלְּקָהּ\hdot
אוֹ מִעֲטַה הַשְּׁחִין\hdot
אוֹ מִחְיַת הַשְּׁחִין\hdot
וְהַמִּכְוָה\hdot
וּמִחְיַת הַמִּכְוָה\hdot
וְהַבֹּהַק\hdot
הָיָה פִסְיוֹן וְהָלַךְ לוֹ פִסְיוֹן\hdot
אוֹ שֶׁהָֽלְכָה לָהּ הָאוֹם\hdot
אוֹ שֶׁנִּתְמַעֲטָה\hdot
וְאֵין בְּזֶה וּבְזֶה כְגָרִיס\hdot
הַשְּׁחִין\hdot
וּמִחְיַת הַשְּׁחִין\hdot
וְהַמִּכְוָה\hdot
וּמִחְיַת הַמִּכְוָה\hdot
וְהַבּוֹהַק\hdot
חוֹלְקִין בֵּין הָאוֹם לַפִּסְיוֹן\hdot
הֲרֵי אֵלּו לְהָקֵל׃}
