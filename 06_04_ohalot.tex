\tractatehead{מסכת אהלות}{ohalot}

\perek{שְׁנַיִם}{טְמֵאִים בַּמֵּת\hdot
אֶחָד טָמֵא טֻמְאַת שִׁבְעָה\hdot
וְאֶחָד טָמֵא טֻמְאַת עֶרֶב\hdot
שְׁלֹשָׁה טְמֵאִין בַּמֵּת\hdot
שְׁנַיִם טְמֵאִין טֻמְאַת שִׁבְעָה\hdot
וְאֶחָד טָמֵא טֻמְאַת עֶרֶב\hdot
אַרְבָּעָה טְמֵאִין בַּמֵּת\hdot
שְׁלֹשָׁה טְמֵאִין טֻמְאַת שִׁבְעָה\hdot
וְאֶחָד טָמֵא טֻמְאַת עֶרֶב\hdot
כֵּיצַד שְׁנַיִם\hdot
אָדָם הַנּוֹגֵעַ בַּמֵּת\hdot
טָמֵא טֻמְאַת שִׁבְעָה\hdot
וְאָדָם הַנּוֹגֵעַ בּוֹ\hdot
טָמֵא טֻמְאַת עָרֶב׃}

\halacha{כֵּיצַד שְׁלֹשָׁה\hdot
כֵּלִים הַנּוֹגְעִים בַּמֵּת\hdot
וְכֵלִים בַּכֵּלִים טְמֵאִין טֻמְאַת שִׁבְעָה\hdot
הַשְּׁלִישִׁי בֵּין אָדָם וּבֵין כֵּלִים\hdot
טְמֵאִים טֻמְאַת עָרֶב׃}

\halacha{כֵּיצַד אַרְבָּעָה\hdot
כֵּלִים הַנּוֹגְעִים בַּמֵּת\hdot
וְאָדָם בַּכֵּלִים\hdot
וְכֵלִים בָּאָדָם\hdot
טְמֵאִין טֻמְאַת שִׁבְעָה\hdot
הָֽרְבִיעִי בֵּין אָדָם וּבֵין כֵּלִים\hdot
טְמֵאִים טֻמְאַת עֶרֶב\hdot
אָמַר רִבִּי עֲקִיבָא יֵשׁ לִי חֲמִישִׁי\hdot
הַשְּׁפוֹד הַתָּחוּב בָּאֹהֶל\hdot
הָאֹהֶל וְהַשְּׁפוֹד\hdot
וְאָדָם הַנּוֹגֵעַ בַּשְּׁפוֹד\hdot
וְכֵלִים בָּאָדָם\hdot
טְמֵאִין טֻמְאַת שִׁבְעָה\hdot
חֲמִישִׁי בֵּין אָדָם בֵּין כֵּלִים\hdot
טְמֵאִים טֻמְאַת עֶרֶב\hdot
אָֽמְרוּ לוֹ אֵין הָאֹהֶל מִתְחַשֵּׁב׃}

\halacha{אָדָם וְכֵלִים מִטַּמְּאִים בַּמֵּת\hdot
חֹמֶר בָּאָדָם מִבַּכֵּלִים\hdot
וּבַכֵּלִים מִבָּאָדָם\hdot
שֶׁהַכֵּלִים שְׁלֹשָׁה\hdot
וְהָאָדָם שְׁנַיִם\hdot
חֹמֶר בָּאָדָם\hdot
שֶׁכָּל זְמָן שֶׁהוּא בָאֶמְצַע\hdot
הֵן אַרְבָּעָה\hdot
וְשֶׁאֵינוֹ בָאֶמְצַע\hdot
הֵן שְׁלֹשָׁה׃}

\halacha{אָדָם וּבְגָדִים מִטַּמְּאִין בַּזָּב\hdot
חֹמֶר בָּאָדָם מִבַּבְּגָדִים\hdot
וּבַבְּגָדִים מִבָּאָדָם\hdot
שֶׁהָאָדָם הַנּוֹגֵעַ בַּזָּב\hdot
מְטַמֵּא בְגָדִים\hdot
וְאֵין בְּגָדִים הַנּוֹגְעִים בַּזָּב\hdot
מְטַמְּאִים בְּגָדִים\hdot
חֹמֶר בַּבְּגָדִים\hdot
שֶׁהַבְּגָדִים הַנּוֹשְׂאִים אֶת הַזָּב מְטַמְּאִים אָדָם\hdot
וְאֵין אָדָם הַנּוֹשֵׂא אֶת הַזָּב מְטַמֵּא אָדָם׃}

\halacha{אָדָם אֵינוֹ מְטַמֵּא עַד שֶׁתֵּצֵא נַפְשׁוֹ\hdot
וַאֲפִילוּ מִגוּיָיד\hdot
וַאֲפִילוּ גוֹסֵס\hdot
זוֹקֵק לַיִּבּוּם וּפוֹטֵר מִן הַיִּבּוּם\hdot
מַאֲכִיל בַּתְּרוּמָה וּפוֹסֵל מִן הַתְּרוּמָה\hdot
וְכֵן בְּהֵמָה וְחַיָּה אֵינָן מְטַמְּאִין עַד שֶׁתֵּצֵא נַפְשָׁם\hdot
הֻתְּזוּ רָאשֵׁיהֶם\hdot
אַף עַל פִּי שֶׁמְּפַרְכְּסִים טְמֵאִין\hdot
כְּגוֹן זָנָב שֶׁל לְטָאָה שֶׁהִיא מְפַרְכֶּסֶת׃}

\halacha{הָאֵיבָרִין אֵין לָהֶן שִׁיעוּר\hdot
אֲפִילוּ פָּחוּת מִכַּזַּיִת מִן הַמֵּת\hdot
וּפָחוּת מִכַּזַּיִת מִן הַנְּבֵלָה\hdot
וּפָחוּת מִכַּעֲדָשָׁה מִן הַשֶּׁרֶץ\hdot
מְטַמְּאִין טֻמְאָתָן׃}

\halacha{מָאתַיִם וְאַרְבָּעִים וּשְׁמוֹנָה אֵבָרִים בָּאָדָם\hdot
שְׁלֹשִׁים בְּפִיסַת הָרֶגֶל\hdot
שִׁשָּׁה בְּכָל אֶצְבַּע\hdot
עֲשָׁרָה בַּקֻּרְסָל\hdot
שְׁנַיִם בַּשּׁוֹק\hdot
חֲמִשָּׁה בָּאַרְכֻּבָּה\hdot
אֶחָד בַּיָּרֵךְ\hdot
שְׁלֹשָׁה בַּקַּטְלִית\hdot
אַחַד עָשָׂר צְלָעוֹת\hdot
שְׁלֹשִׁים בְּפִיסַת הַיָּד\hdot
שִׁשָּׁה בְּכָל אֶצְבַּע\hdot
שְׁנַיִם בַּקָּנֶה\hdot
וּשְׁנַיִם בַּמַּרְפֵּק\hdot
אֶחָד בַּזְּרוֹעַ\hdot
וְאַרְבָּעָה בַּכָּתֵף\hdot
מֵאָה וְאֶחָד מִזֶּה\hdot
וּמֵאָה וְאֶחָד מִזֶּה\hdot
וּשְׁמוֹנָה עָשָׂר חֻלְיוֹת בַּשִּׁדְרָה\hdot
תִּשְׁעָה בָּרֹאשׁ\hdot
שְׁמוֹנָה בַּצַּוָּאר\hdot
שִׁשָּׁה בַּמַּפְתֵּחַ שֶׁל לֵב\hdot
וַחֲמִשָּׁה בִּנְקָבָיו\hdot
כָּל אֶחָד וְאֶחָד\hdot
מְטַמֵּא\hdot
בְּמַגַּע וּבְמַשָּׂא וּבְאֹהֶל\hdot
אֵימָתַי בִּזְמָן שֶׁיֵּשׁ עֲלֵיהֶם בָּשָׂר כָּרָאוּי\hdot
אֲבָל אִם אֵין עֲלֵיהֶם בָּשָׂר כָּרָאוּי\hdot
מְטַמְּאִין בְּמַגַּע וּבְמַשָּׂא\hdot
וְאֵינָן מְטַמְּאִין בְּאֹהֶל׃}
